cmake_minimum_required(VERSION 3.16)
project(QtFreeImage VERSION 1.0 LANGUAGES C CXX)

set(CMAKE_CXX_STANDARD 11)

option(UseTIFFLib "Enable using TIFF library" OFF)

if(EMSCRIPTEN)
    set(CMAKE_CXX_FLAGS ${CMAKE_CXX_FLAGS} "-s USE_PTHREADS")
    set(CMAKE_C_FLAGS ${CMAKE_C_FLAGS} "-s USE_PTHREADS")
endif()
add_subdirectory(LibPNG)
add_subdirectory(Examples)
add_subdirectory(LibRawLite)
add_subdirectory(FreeImage)
add_subdirectory(FreeImageToolkit)
add_subdirectory(MetaData)
add_subdirectory(LibJpeg)
add_subdirectory(LibJXR)
if(UseTIFFLib)
    add_subdirectory(LibTIFF4)
endif()
add_subdirectory(ZLib)
add_subdirectory(LibOpenJPEG)
add_subdirectory(LibWebP)
add_subdirectory(OpenEXR)
